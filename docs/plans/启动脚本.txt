从upstream merge完验证命令
# 1. 切到 master，同步官方最新
git checkout master
git fetch upstream
git merge upstream/master

# 2. 切到开发分支，合并 master
git checkout feat/enterprise-platform
git merge master

# 3. 如有冲突：
#    pnpm-lock.yaml 直接用官方版:
#      git checkout --theirs pnpm-lock.yaml
#      git add pnpm-lock.yaml
#    其他共享文件以官方为准，重新套企业改动
#    全部解决后: git add . && git merge --continue

# 4. 验证
pnpm install
pnpm run typecheck
pnpm run build
pnpm dsh --profile enterprise

# 5. 只推 feat 到你的 fork
git push origin feat/enterprise-platform
